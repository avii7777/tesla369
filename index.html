// آماده‌سازی ابزار تلگرام const tg = window.Telegram.WebApp; tg.expand(); // باز شدن تمام صفحه // خواندن نام کاربر واقعی از تلگرام if (tg.initDataUnsafe && tg.initDataUnsafe.user) { document.getElementById('username').innerText = tg.initDataUnsafe.user.first_name; } else { document.getElementById('username').innerText = "کاربر تسلا"; } // افکت لمس گوی جادویی function triggerLuckEffect() { tg.HapticFeedback.impactOccurred('medium'); // ویبره ظریف گوشی // ارسال دستور شانس به ربات tg.sendData(JSON.stringify({ action: 'spin_wheel', timestamp: Date.now() })); // بستن خودکار مینی‌اپ بعد از ارسال دیتا tg.close(); } // عملکرد دکمه اتاق‌های بازی function openRooms() { tg.HapticFeedback.impactOccurred('light'); // ارسال درخواست به ربات برای مدیریت یا مشاهده اتاق‌ها tg.sendData(JSON.stringify({ action: 'view_rooms' })); tg.close(); } // عملکرد دکمه بخت‌آزمایی شبانه function openNightly() { tg.HapticFeedback.impactOccurred('light'); tg.sendData(JSON.stringify({ action: 'buy_ticket' })); tg.close(); } // متصل کردن دکمه‌های HTML به این توابع جدید // (اینجا دکمه‌ها را به توابع جاوااسکریپت وصل می‌کنیم) document.querySelectorAll('button')[0].setAttribute('onclick', 'openNightly()'); document.querySelectorAll('button')[1].setAttribute('onclick', 'openRooms()');
